function SectionCard({ title, icon: Icon, children, headerAction, color = "blue", className = "" }: any) {
  const colors: Record<string, string> = {
    blue: "border-blue-200 text-blue-700 bg-blue-50/50",
    green: "border-green-200 text-green-700 bg-green-50/50",
    purple: "border-purple-200 text-purple-700 bg-purple-50/50",
    orange: "border-orange-200 text-orange-700 bg-orange-50/50",
    red: "border-red-200 text-red-700 bg-red-50/50",
    slate: "border-slate-200 text-slate-700 bg-slate-50/50",
  }
  const colorClass = colors[color] || colors.blue

  return (
    <Card className={`overflow-hidden border border-slate-200 shadow-sm ${className}`}>
      <CardHeader className="p-0 border-b bg-slate-50/50">
        <div className="flex items-center justify-between px-4 py-3">
          <div className="flex items-center gap-2">
            <Icon className={`h-4 w-4 ${colorClass.split(" ")[1]}`} />
            <h3 className="font-semibold text-sm text-slate-700">{title}</h3>
          </div>
          {headerAction && <div>{headerAction}</div>}
        </div>
      </CardHeader>
      <CardContent className="p-4">{children}</CardContent>
    </Card>
  )
}

              </div>
              <div>

                <p className="text-sm font-medium">{admission.bed.ward.type}</p>

                <p className="text-sm">{admission.reason || "N/A"}</p>
              </div>
              <div>
                <p className="text-[10px] font-semibold tracking-wider text-slate-500 uppercase mb-1">Past Medical History</p>
                <p className="text-sm">{patient.pastMedicalHistory || patient.medicalHistory || "None"}</p>

                </div>
              )}
            </div>
          </SectionCard>

             <div className="flex items-center justify-between">
                <p className="text-sm text-slate-500">View patient documents in the Notes tab.</p>

                  Assign
                </Button>
              )
            }
          >

          )}

          <SectionCard title="Insurance / Sponsor" icon={Shield} color="blue">

              </div>
            )}
          </SectionCard>

          {/* LATEST VITALS */}
          <SectionCard title="Latest Vitals" icon={Activity} color="red">
            <div className="text-center py-6">

            </div>

          <SectionCard title="Payment Information" icon={CreditCard} color="orange">
            <div className="space-y-3">
              <div className="flex justify-between text-sm">

                <span>₹{(admission.bed.pricePerDay * daysAdmitted).toLocaleString()}</span>
              </div>
              <div className="flex justify-between text-sm font-medium pt-3 border-t border-slate-100">
                <span>Total Bill</span>
                <span>₹{admission.totalBillAmount.toLocaleString()}</span>
              </div>
              <div className="flex justify-between text-sm text-green-600">
                <span>Advance Deposit</span>
                <span>-₹{admission.initialDeposit.toLocaleString()}</span>
              </div>
              <div className="flex justify-between text-base font-bold pt-3 border-t border-slate-200 text-red-600">
                <span>Balance Due</span>
                <span>₹{(admission.totalBillAmount - admission.initialDeposit).toLocaleString()}</span>
              </div>
            </div>
          </SectionCard>

          {/* BED TRANSFER HISTORY */}
          <SectionCard title="Bed Transfer History" icon={BedDouble} color="blue" headerAction={<Button variant="ghost" size="sm" className="h-7 text-xs border border-blue-200 text-blue-700 bg-white">Transfer Bed</Button>}>
             <div className="text-center py-4 text-xs text-slate-500">
               <BedDouble className="h-6 w-6 text-slate-300 mx-auto mb-1" />
               <p>No bed transfers</p>
               <p className="text-[10px] mt-1">Patient has been in the same bed since admission.</p>
             </div>
          </SectionCard>

        </div>
      </div>
    </div>
  )
}
